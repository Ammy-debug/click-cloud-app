/* DIRECT ANDROID PDF DOWNLOAD & SAVE */ async function downloadPdfDirectly() { const btn = document.getElementById('modal-btn-download'); const origText = btn.innerText; btn.innerText = "⏳ Saving..."; let custName = document.getElementById('cust-name').value.trim() || "Client"; let fileName = `Quotation_${custName.replace(/\s+/g, '_')}_${Date.now()}.pdf`; try { const canvas = await createQuotationCanvas(); const imgData = canvas.toDataURL('image/jpeg', 0.95); const { jsPDF } = window.jspdf; const pdf = new jsPDF('p', 'mm', 'a4'); const pdfWidth = pdf.internal.pageSize.getWidth(); const pdfHeight = (canvas.height * pdfWidth) / canvas.width; pdf.addImage(imgData, 'JPEG', 0, 0, pdfWidth, pdfHeight); const base64Pdf = pdf.output('datauristring').split(',')[1]; // 1. Native Capacitor File Saving & Prompt if (window.Capacitor && window.Capacitor.Plugins) { const { Filesystem, Share } = window.Capacitor.Plugins; if (Filesystem) { const savedFile = await Filesystem.writeFile({ path: fileName, data: base64Pdf, directory: 'DOCUMENTS', recursive: true }).catch(async () => { return await Filesystem.writeFile({ path: fileName, data: base64Pdf, directory: 'CACHE' }); }); btn.innerText = origText; if (Share) { await Share.share({ title: fileName, text: `Quotation PDF saved for ${custName}`, url: savedFile.uri, dialogTitle: 'Save / Open PDF' }); } else { alert("✓ PDF Successfully Saved to Documents: " + fileName); } return; } } // 2. Web Share API Fallback const pdfBlob = pdf.output('blob'); const pdfFile = new File([pdfBlob], fileName, { type: 'application/pdf' }); if (navigator.canShare && navigator.canShare({ files: [pdfFile] })) { await navigator.share({ files: [pdfFile], title: fileName, text: `Official Quotation for ${custName}` }); btn.innerText = origText; return; } // 3. Direct Browser Anchor Fallback const downloadUrl = URL.createObjectURL(pdfBlob); const a = document.createElement('a'); a.href = downloadUrl; a.download = fileName; document.body.appendChild(a); a.click(); document.body.removeChild(a); setTimeout(() => URL.revokeObjectURL(downloadUrl), 1500); btn.innerText = origText; alert("✓ PDF Download Complete!"); } catch(err) { console.error("PDF download error:", err); btn.innerText = origText; alert("PDF Download trigger ho gaya!"); } }
